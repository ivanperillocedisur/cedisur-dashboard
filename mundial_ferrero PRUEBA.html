} else if (tipo.includes('eliminar')) { eliminarFila(sheetBase, vendedor, diaActual, codigo); } else { throw new Error('Tipo no reconocido: ' + tipo); } sheetSol.getRange(fila, COL_SOL.ESTADO).setValue('Aprobado'); marcarAplicado(sheetSol, fila); return ContentService .createTextOutput(JSON.stringify({ ok: true })) .setMimeType(ContentService.MimeType.JSON); } catch (err) { marcarError(sheetSol, fila, err.message); return ContentService .createTextOutput(JSON.stringify({ ok: false, error: err.message })) .setMimeType(ContentService.MimeType.JSON); } } /** * El panel de supervisor rechaza UNA solicitud puntual, con motivo opcional. */ function rechazarSolicitudPanel(ss, data) { const sheetSol = ss.getSheetByName(HOJA_SOLICITUDES); const zonaVendedor = zonaDeVendedor(ss, data.vendedor); if (!zonaVendedor || zonaVendedor !== norm_(data.zonaSupervisor)) { return ContentService .createTextOutput(JSON.stringify({ ok: false, error: 'El vendedor no pertenece a tu zona' })) .setMimeType(ContentService.MimeType.JSON); ': ' + motivo : '')) .setBackground('#F1F1F1') .setFontColor('#6B6760') .setFontWeight('bold'); return ContentService .createTextOutput(JSON.stringify({ ok: true })) .setMimeType(ContentService.MimeType.JSON); } /** * Busca el cliente en BASE RUTEO por código y escribe Latitud/Longitud en col I y J. * Si hay múltiples filas con el mismo código (cliente en varios días), actualiza todas. */ function guardarCoordenadas(ss, data) { const sheet = ss.getSheetByName(HOJA_BASE); } // ═══════════════════════════════════════════════════════════════════════════ // FUNCIÓN 3 — Aplicar solicitudes pendientes (a demanda) // ═══════════════════════════════════════════════════════════════════════════ function aplicarSolicitudesPendientes() { const ss = SpreadsheetApp.openById(SHEET_ID); const sheetSol = ss.getSheetByName(HOJA_SOLICITUDES); const sheetBase = ss.getSheetByName(HOJA_BASE); if (!sheetSol || !sheetBase) { SpreadsheetApp.getUi().alert('Error: no se encontraron las hojas necesarias.'); const resultado = String(row[COL_SOL.RESULTADO - 1] || '').trim(); if (estado !== 'PENDIENTE' || resultado.includes('APLICADO')) continue; const fila = i + 1; const vendedor = String(row[COL_SOL.VENDEDOR - 1]).trim(); } else if (tipo.includes('día') || tipo.includes('dia')) { const matchDia = detalle.match(/Nuevo d[íi]a solicitado:\s*([A-ZÁÉÍÓÚÑ]+)/i); const matchOrden = detalle.match(/Nuevo orden:\s*(\d+)/i); if (!matchDia) throw new Error('No se encontró nuevo día en: ' + detalle); } else if (tipo.includes('eliminar')) { eliminarFila(sheetBase, vendedor, diaActual, codigo); } else { throw new Error('Tipo no reconocido: ' + tipo); } sheetSol.getRange(fila, COL_SOL.ESTADO).setValue('Aprobado'); marcarAplicado(sheetSol, fila); procesadas++; } catch (err) { marcarError(sheetSol, fila, err.message); errores++; Logger.log('Error en fila ' + fila + ': ' + err.message); } } SpreadsheetApp.getUi().alert( '✅ Proceso finalizado\n\n' + 'Solicitudes aplicadas: ' + procesadas + '\n' + 'Errores: ' + errores ); } // ═══════════════════════════════════════════════════════════════════════════ // LECTURA DE BLOQUE CON FÓRMULAS // Lee valores Y fórmulas de cada fila del bloque para preservarlas // ═══════════════════════════════════════════════════════════════════════════ function leerBloque(sheet, vendedor, dia) { const numCols = sheet.getLastColumn(); const lastRow = sheet.getLastRow(); const valores = sheet.getRange(1, 1, lastRow, numCols).getValues(); const formulas = sheet.getRange(1, 1, lastRow, numCols).getFormulas(); var bloque = []; for (var i = 1; i < valores.length; i++) { var v = String(valores[i][COL_BASE.VENDEDOR - 1]).trim(); var d = String(valores[i][COL_BASE.DIA - 1]).trim().toUpperCase(); if (v === vendedor && d === dia) { bloque.push({ rowIndex: i + 1, codigo: String(valores[i][COL_BASE.CODIGO - 1]).trim(), orden: parseInt(valores[i][COL_BASE.ORDEN - 1]) || 0, valores: valores[i].slice(), formulas: formulas[i].slice() }); } } return bloque; } // ═══════════════════════════════════════════════════════════════════════════ // ESCRITURA DE BLOQUE PRESERVANDO FÓRMULAS EN B, C, D // ═══════════════════════════════════════════════════════════════════════════ function escribirBloque(sheet, bloque, primeraFila) { const numCols = sheet.getLastColumn(); for (var j = 0; j < bloque.length; j++) { var filaDestino = primeraFila + j; var vals = bloque[j].valores.slice(); var fmls = bloque[j].formulas.slice(); // Actualizar número de orden vals[COL_BASE.ORDEN - 1] = j + 1; // Escribir columna por columna: fórmula si existe, valor si no for (var col = 0; col < numCols; col++) { var celda = sheet.getRange(filaDestino, col + 1); if (COLS_CON_FORMULA.indexOf(col) !== -1 && fmls[col] !== '') { // Ajustar la referencia de fila en la fórmula var formulaAjustada = ajustarFilaEnFormula(fmls[col], bloque[j].rowIndex, filaDestino); celda.setFormula(formulaAjustada); } else { celda.setValue(vals[col]); } } } } // ═══════════════════════════════════════════════════════════════════════════ // AJUSTE DE REFERENCIAS DE FILA EN FÓRMULAS // Reemplaza referencias absolutas a la fila original por la fila destino // ═══════════════════════════════════════════════════════════════════════════ function ajustarFilaEnFormula(formula, filaOrigen, filaDestino) { if (filaOrigen === filaDestino) return formula; // Reemplaza referencias del tipo A2, B2, $A2, A$2, $A$2, etc. var regex = new RegExp('(\\$?[A-Z]+\\$?)' + filaOrigen + '(?![0-9])', 'g'); return formula.replace(regex, '$1' + filaDestino); } // ═══════════════════════════════════════════════════════════════════════════ // OPERACIONES FÍSICAS SOBRE BASE RUTEO // ═══════════════════════════════════════════════════════════════════════════ function moverFilaOrden(sheet, vendedor, dia, codigo, nuevoOrden) { var bloque = leerBloque(sheet, vendedor, dia); if (!bloque.length) throw new Error('Sin clientes para ' + vendedor + ' el ' + dia); var objetivo = bloque.find(function(f) { return f.codigo === codigo; }); if (!objetivo) throw new Error('Cliente ' + codigo + ' no encontrado en ' + dia + ' (' + vendedor + ')'); // Reordenar: sacar al cliente y reinsertarlo en la posición solicitada bloque.sort(function(a, b) { return a.orden - b.orden; }); var sinObjetivo = bloque.filter(function(f) { return f.codigo !== codigo; }); var pos = Math.min(nuevoOrden - 1, sinObjetivo.length); sinObjetivo.splice(pos, 0, objetivo); var primeraFila = bloque.reduce(function(min, f) { return f.rowIndex < min ? f.rowIndex : min; }, bloque[0].rowIndex); // Borrar filas del bloque de abajo a arriba bloque.slice() .sort(function(a, b) { return b.rowIndex - a.rowIndex; }) .forEach(function(f) { sheet.deleteRow(f.rowIndex); }); // Insertar filas vacías y escribir con fórmulas preservadas sheet.insertRowsBefore(primeraFila, sinObjetivo.length); escribirBloque(sheet, sinObjetivo, primeraFila); } function moverFilaDia(sheet, vendedor, diaAntiguo, codigo, diaNuevo, nuevoOrden) { const numCols = sheet.getLastColumn(); const lastRow = sheet.getLastRow(); const valores = sheet.getRange(1, 1, lastRow, numCols).getValues(); var filaCliente = -1; for (var i = 1; i < valores.length; i++) { var v = String(valores[i][COL_BASE.VENDEDOR - 1]).trim(); var d = String(valores[i][COL_BASE.DIA - 1]).trim().toUpperCase(); var c = String(valores[i][COL_BASE.CODIGO - 1]).trim(); if (v === vendedor && d === diaAntiguo && c === codigo) { filaCliente = i + 1; break; } } if (filaCliente === -1) throw new Error('Cliente ' + codigo + ' no encontrado en ' + diaAntiguo); // Actualizar DIA y poner orden temporal alto sheet.getRange(filaCliente, COL_BASE.DIA).setValue(diaNuevo); sheet.getRange(filaCliente, COL_BASE.ORDEN).setValue(9999); // Reordenar físicamente el día antiguo reordenarBloqueF(sheet, vendedor, diaAntiguo); // Insertar en la posición correcta del día nuevo moverFilaOrden(sheet, vendedor, diaNuevo, codigo, nuevoOrden); } function eliminarFila(sheet, vendedor, dia, codigo) { const lastRow = sheet.getLastRow(); const valores = sheet.getRange(1, 1, lastRow, sheet.getLastColumn()).getValues(); var filaCliente = -1; for (var i = 1; i < valores.length; i++) { var v = String(valores[i][COL_BASE.VENDEDOR - 1]).trim(); var d = String(valores[i][COL_BASE.DIA - 1]).trim().toUpperCase(); var c = String(valores[i][COL_BASE.CODIGO - 1]).trim(); if (v === vendedor && d === dia && c === codigo) { filaCliente = i + 1; break; } } if (filaCliente === -1) throw new Error('Cliente ' + codigo + ' no encontrado en ' + dia); sheet.deleteRow(filaCliente); reordenarBloqueF(sheet, vendedor, dia); } function reordenarBloqueF(sheet, vendedor, dia) { var bloque = leerBloque(sheet, vendedor, dia); if (!bloque.length) return; bloque.sort(function(a, b) { return a.orden - b.orden; }); var primeraFila = bloque.reduce(function(min, f) { return f.rowIndex < min ? f.rowIndex : min; }, bloque[0].rowIndex); // Borrar de abajo a arriba bloque.slice() .sort(function(a, b) { return b.rowIndex - a.rowIndex; }) .forEach(function(f) { sheet.deleteRow(f.rowIndex); }); // Reinsertar con fórmulas preservadas sheet.insertRowsBefore(primeraFila, bloque.length); escribirBloque(sheet, bloque, primeraFila); } // ═══════════════════════════════════════════════════════════════════════════ // MARCADO EN HOJA SOLICITUDES // ═══════════════════════════════════════════════════════════════════════════ function marcarAplicado(sheet, fila) { sheet.getRange(fila, COL_SOL.RESULTADO) .setValue('✅ APLICADO') .setBackground('#B7F5C8') .setFontColor('#1A5C2A') .setFontWeight('bold'); } function marcarError(sheet, fila, mensaje) { sheet.getRange(fila, COL_SOL.RESULTADO) .setValue('❌ ERROR: ' + mensaje) .setBackground('#FDEAEA') .setFontColor('#A02020') .setFontWeight('bold'); }
